# lanbox Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建 lanbox —— 一个零依赖的 Go 单二进制服务，让两台电脑通过内嵌 Web 页面共享文本与文件（上传、查看、下载、标记已取走、删除），HTTPS 单端口、独立账号、SSE 实时刷新。

**Architecture:** 单个 Go 进程，分层为 `config`（配置）、`tlsutil`（自签证书）、`store`（内容与附件的文件系统存储 + 磁盘守卫）、`auth`（PBKDF2 用户库 + 内存会话 + 登录限速）、`httpapi`（路由/中间件/流式上传/SSE）。前端为原生 ES module 单页，经 `//go:embed` 打进二进制，无构建步骤。

**Tech Stack:** Go 1.26（**仅标准库**，无第三方依赖）、`net/http`（Go 1.22+ 路由模式 `GET /api/items/{id}`）、`crypto/pbkdf2`、`crypto/tls` + `crypto/x509`、`embed`、原生 JS/CSS。

**规格来源：** `docs/superpowers/specs/2026-11-26-lanbox-design.md`（实现时逐节对照）

**模块边界规则**（实现时不得违反）：
- `store` 不导入 `net/http`、不认识 multipart，只接受 `io.Reader`
- `httpapi` 不拼接任何磁盘路径，一律经 `store` 方法
- `auth` 不导入 `httpapi`，会话校验以 `Lookup(token) (*User, bool)` 暴露
- `config`、`tlsutil` 不依赖仓库内其它包
- 每个文件保持单一职责；超过 ~300 行时按上表职责再拆

// Package config 负责读取 lanbox 的配置：默认值 → config.json → 环境变量 → 校验。
package config

import (
	"encoding/json"
	"errors"
	"fmt"
	"os"
	"strconv"
	"time"
)

// TLS 是证书路径配置。两者都为空表示自动生成自签证书。
type TLS struct {
	CertFile string `json:"cert_file"`
	KeyFile  string `json:"key_file"`
}

// Config 是 lanbox 的全部配置项。
type Config struct {
	Listen          string `json:"listen"`
	DataDir         string `json:"data_dir"`
	TLS             TLS    `json:"tls"`
	MaxFileSizeMB   int64  `json:"max_file_size_mb"`
	MinFreeDiskMB   int64  `json:"min_free_disk_mb"`
	MaxTextBytes    int    `json:"max_text_bytes"`
	SessionTTLHours int    `json:"session_ttl_hours"`
}

// Defaults 返回可直接使用的默认配置。
func Defaults() Config {
	return Config{
		Listen:          ":8443",
		DataDir:         "./data",
		MaxFileSizeMB:   1024,
		MinFreeDiskMB:   1024,
		MaxTextBytes:    65536,
		SessionTTLHours: 168,
	}
}

// Load 读取 path（不存在则仅用默认值），再应用环境变量并校验。
func Load(path string) (Config, error) {
	c := Defaults()
	raw, err := os.ReadFile(path)
	switch {
	case err == nil:
		if err := json.Unmarshal(raw, &c); err != nil {
			return c, fmt.Errorf("解析配置文件 %s 失败: %w", path, err)
		}
	case errors.Is(err, os.ErrNotExist):
		// 允许没有配置文件
	default:
		return c, fmt.Errorf("读取配置文件 %s 失败: %w", path, err)
	}
	if err := applyEnv(&c); err != nil {
		return c, err
	}
	if err := c.Validate(); err != nil {
		return c, err
	}
	return c, nil
}

func applyEnv(c *Config) error {
	if v, ok := os.LookupEnv("LANBOX_LISTEN"); ok {
		c.Listen = v
	}
	if v, ok := os.LookupEnv("LANBOX_DATA_DIR"); ok {
		c.DataDir = v
	}
	if v, ok := os.LookupEnv("LANBOX_TLS_CERT_FILE"); ok {
		c.TLS.CertFile = v
	}
	if v, ok := os.LookupEnv("LANBOX_TLS_KEY_FILE"); ok {
		c.TLS.KeyFile = v
	}
	ints := []struct {
		env string
		dst *int64
	}{
		{"LANBOX_MAX_FILE_SIZE_MB", &c.MaxFileSizeMB},
		{"LANBOX_MIN_FREE_DISK_MB", &c.MinFreeDiskMB},
	}
	for _, it := range ints {
		v, ok := os.LookupEnv(it.env)
		if !ok {
			continue
		}
		n, err := strconv.ParseInt(v, 10, 64)
		if err != nil {
			return fmt.Errorf("环境变量 %s 不是整数: %q", it.env, v)
		}
		*it.dst = n
	}
	if v, ok := os.LookupEnv("LANBOX_MAX_TEXT_BYTES"); ok {
		n, err := strconv.Atoi(v)
		if err != nil {
			return fmt.Errorf("环境变量 LANBOX_MAX_TEXT_BYTES 不是整数: %q", v)
		}
		c.MaxTextBytes = n
	}
	if v, ok := os.LookupEnv("LANBOX_SESSION_TTL_HOURS"); ok {
		n, err := strconv.Atoi(v)
		if err != nil {
			return fmt.Errorf("环境变量 LANBOX_SESSION_TTL_HOURS 不是整数: %q", v)
		}
		c.SessionTTLHours = n
	}
	return nil
}

// Validate 校验配置自洽性，返回面向使用者的中文错误。
func (c Config) Validate() error {
	if c.Listen == "" {
		return errors.New("listen 不能为空，例如 \":8443\"")
	}
	if c.DataDir == "" {
		return errors.New("data_dir 不能为空")
	}
	if c.MaxFileSizeMB <= 0 {
		return errors.New("max_file_size_mb 必须大于 0")
	}
	if c.MinFreeDiskMB < 0 {
		return errors.New("min_free_disk_mb 不能为负数")
	}
	if c.MaxTextBytes <= 0 {
		return errors.New("max_text_bytes 必须大于 0")
	}
	if c.SessionTTLHours <= 0 {
		return errors.New("session_ttl_hours 必须大于 0")
	}
	if (c.TLS.CertFile == "") != (c.TLS.KeyFile == "") {
		return errors.New("tls.cert_file 与 tls.key_file 必须同时配置或同时留空")
	}
	return nil
}

// MaxFileBytes 是单文件大小上限（字节）。
func (c Config) MaxFileBytes() int64 { return c.MaxFileSizeMB << 20 }

// MinFreeBytes 是必须保留的空闲磁盘下限（字节）。
func (c Config) MinFreeBytes() int64 { return c.MinFreeDiskMB << 20 }

// SessionTTL 是会话有效期。
func (c Config) SessionTTL() time.Duration {
	return time.Duration(c.SessionTTLHours) * time.Hour
}
```

// Package tlsutil 负责确定 lanbox 使用的证书：优先用户配置，否则生成并复用自签证书。
package tlsutil

import (
	"crypto/ecdsa"
	"crypto/elliptic"
	"crypto/rand"
	"crypto/tls"
	"crypto/x509"
	"crypto/x509/pkix"
	"encoding/pem"
	"errors"
	"fmt"
	"math/big"
	"net"
	"os"
	"path/filepath"
	"strings"
	"time"
)

const selfSignedValidity = 10 * 365 * 24 * time.Hour

// Ensure 返回应当使用的证书与私钥路径。
// certFile/keyFile 均非空：校验后直接使用；均为空：复用或生成 dataDir/tls/{cert.pem,key.pem}。
// generated 表示本次是否新生成。
func Ensure(dataDir, certFile, keyFile string, hosts []string) (certPath, keyPath string, generated bool, err error) {
	if (certFile == "") != (keyFile == "") {
		return "", "", false, errors.New("tls.cert_file 与 tls.key_file 必须同时配置或同时留空")
	}
	if certFile != "" {
		if _, err := tls.LoadX509KeyPair(certFile, keyFile); err != nil {
			return "", "", false, fmt.Errorf("加载证书失败（cert=%s key=%s）: %w", certFile, keyFile, err)
		}
		return certFile, keyFile, false, nil
	}

	dir := filepath.Join(dataDir, "tls")
	certPath = filepath.Join(dir, "cert.pem")
	keyPath = filepath.Join(dir, "key.pem")
	if fileExists(certPath) && fileExists(keyPath) {
		if _, err := tls.LoadX509KeyPair(certPath, keyPath); err == nil {
			return certPath, keyPath, false, nil
		}
		// 已存在的证书坏了：重新生成
	}
	if err := os.MkdirAll(dir, 0o700); err != nil {
		return "", "", false, fmt.Errorf("创建证书目录 %s 失败: %w", dir, err)
	}
	certPEM, keyPEM, err := generate(hosts, selfSignedValidity)
	if err != nil {
		return "", "", false, err
	}
	if err := os.WriteFile(certPath, certPEM, 0o644); err != nil {
		return "", "", false, fmt.Errorf("写入证书失败: %w", err)
	}
	if err := os.WriteFile(keyPath, keyPEM, 0o600); err != nil {
		return "", "", false, fmt.Errorf("写入私钥失败: %w", err)
	}
	return certPath, keyPath, true, nil
}

func generate(hosts []string, valid time.Duration) (certPEM, keyPEM []byte, err error) {
	priv, err := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
	if err != nil {
		return nil, nil, fmt.Errorf("生成私钥失败: %w", err)
	}
	serial, err := rand.Int(rand.Reader, new(big.Int).Lsh(big.NewInt(1), 128))
	if err != nil {
		return nil, nil, fmt.Errorf("生成序列号失败: %w", err)
	}
	tmpl := &x509.Certificate{
		SerialNumber:          serial,
		Subject:               pkix.Name{CommonName: "lanbox", Organization: []string{"lanbox"}},
		NotBefore:             time.Now().Add(-time.Hour),
		NotAfter:              time.Now().Add(valid),
		KeyUsage:              x509.KeyUsageDigitalSignature | x509.KeyUsageKeyEncipherment,
		ExtKeyUsage:           []x509.ExtKeyUsage{x509.ExtKeyUsageServerAuth},
		BasicConstraintsValid: true,
	}
	for _, h := range hosts {
		h = strings.TrimSpace(h)
		if h == "" {
			continue
		}
		if ip := net.ParseIP(h); ip != nil {
			tmpl.IPAddresses = append(tmpl.IPAddresses, ip)
		} else {
			tmpl.DNSNames = append(tmpl.DNSNames, h)
		}
	}
	der, err := x509.CreateCertificate(rand.Reader, tmpl, tmpl, &priv.PublicKey, priv)
	if err != nil {
		return nil, nil, fmt.Errorf("签发证书失败: %w", err)
	}
	keyDER, err := x509.MarshalECPrivateKey(priv)
	if err != nil {
		return nil, nil, fmt.Errorf("序列化私钥失败: %w", err)
	}
	certPEM = pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: der})
	keyPEM = pem.EncodeToMemory(&pem.Block{Type: "EC PRIVATE KEY", Bytes: keyDER})
	return certPEM, keyPEM, nil
}

// Hosts 汇总证书应包含的主机名/IP：localhost、回环、本机各网卡 IP、主机名、listen 中的主机部分。
func Hosts(listen string) []string {
	seen := map[string]bool{}
	var out []string
	add := func(h string) {
		h = strings.TrimSpace(h)
		if h == "" || seen[h] {
			return
		}
		seen[h] = true
		out = append(out, h)
	}
	add("localhost")
	add("127.0.0.1")
	add("::1")
	if host, _, err := net.SplitHostPort(listen); err == nil {
		if ip := net.ParseIP(host); ip != nil && !ip.IsUnspecified() {
			add(host)
		} else if ip == nil && host != "" {
			add(host)
		}
	}
	if hn, err := os.Hostname(); err == nil {
		add(hn)
	}
	if ips, err := LocalIPs(); err == nil {
		for _, ip := range ips {
			add(ip)
		}
	}
	return out
}

// LocalIPs 返回本机所有非未指定地址（含回环）。
func LocalIPs() ([]string, error) {
	addrs, err := net.InterfaceAddrs()
	if err != nil {
		return nil, err
	}
	var out []string
	for _, a := range addrs {
		ipNet, ok := a.(*net.IPNet)
		if !ok || ipNet.IP.IsUnspecified() {
			continue
		}
		out = append(out, ipNet.IP.String())
	}
	return out, nil
}

		if strings.Contains(strings.ToLower(f.Name), needleLower) {

// Store 是内容存储。itemsRoot 是 items 目录（data/items）。
type Store struct {
	itemsRoot string
	opts      Options
	mu        sync.RWMutex
	index     map[string]*Item
}

// Open 创建目录结构并扫描已有内容到内存索引。
func Open(itemsRoot string, opts Options) (*Store, error) {
	if opts.FreeSpace == nil {
		opts.FreeSpace = diskFree
	}
	if err := os.MkdirAll(filepath.Join(itemsRoot), 0o750); err != nil {
		return nil, fmt.Errorf("创建数据目录 %s 失败: %w", itemsRoot, err)
	}
	s := &Store{itemsRoot: itemsRoot, opts: opts, index: map[string]*Item{}}
	entries, err := os.ReadDir(itemsRoot)
	if err != nil {
		return nil, fmt.Errorf("读取数据目录失败: %w", err)
	}
	for _, e := range entries {
		if !e.IsDir() {
			continue
		}
		id := e.Name()
		it, err := readMeta(filepath.Join(itemsRoot, id, "meta.json"))
		if err != nil {
			log.Printf("warn: 跳过损坏的内容 %s: %v", id, err)
			continue
		}
		if it.ID == "" {
			it.ID = id
		}
		if it.Files == nil {
			it.Files = []File{}
		}
		s.index[it.ID] = it
	}
	return s, nil
}

func (s *Store) itemDir(id string) string { return filepath.Join(s.itemsRoot, id) }

func (s *Store) filesDir(id string) string { return filepath.Join(s.itemsRoot, id, "files") }

func readMeta(path string) (*Item, error) {
	raw, err := os.ReadFile(path)
	if err != nil {
		return nil, err
	}
	var it Item
	if err := json.Unmarshal(raw, &it); err != nil {
		return nil, err
	}
	return &it, nil
}

// saveMeta 原子写 meta.json：临时文件 → fsync → rename。调用方必须持有写锁。
func (s *Store) saveMeta(it *Item) error {
	dir := s.itemDir(it.ID)
	raw, err := json.MarshalIndent(it, "", "  ")
	if err != nil {
		return fmt.Errorf("序列化元数据失败: %w", err)
	}
	tmp := filepath.Join(dir, "meta.json.tmp")
	f, err := os.OpenFile(tmp, os.O_WRONLY|os.O_CREATE|os.O_TRUNC, 0o640)
	if err != nil {
		return fmt.Errorf("写入元数据失败: %w", err)
	}
	if _, err := f.Write(raw); err != nil {
		f.Close()
		os.Remove(tmp)
		return fmt.Errorf("写入元数据失败: %w", err)
	}
	if err := f.Sync(); err != nil {
		f.Close()
		os.Remove(tmp)
		return fmt.Errorf("同步元数据失败: %w", err)
	}
	if err := f.Close(); err != nil {
		os.Remove(tmp)
		return fmt.Errorf("关闭元数据失败: %w", err)
	}
	if err := os.Rename(tmp, filepath.Join(dir, "meta.json")); err != nil {
		os.Remove(tmp)
		return fmt.Errorf("替换元数据失败: %w", err)
	}
	return nil
}

// NewID 生成按时间有序的 ID：13 位毫秒时间戳 + 8 位随机十六进制。
func NewID() string {
	var b [4]byte
	for i := range b {
		b[i] = byte(rand.UintN(256))
	}
	return fmt.Sprintf("%013d-%s", time.Now().UnixMilli(), hex.EncodeToString(b[:]))
}

package store

import (
	"fmt"
	"os"
	"sort"
	"strings"
	"time"

// CreateInput 是新建内容的输入（附件随后通过 AddFile 追加）。
type CreateInput struct {
	Text      string
	CreatedBy string
	SourceIP  string
}

// ListQuery 是列表/搜索参数。
type ListQuery struct {
	Limit  int
	Before string
	Query  string
	Todo   bool
}

// ListResult 是分页结果。NextCursor 为空表示没有更多。
type ListResult struct {
	Items      []*Item `json:"items"`
	NextCursor string  `json:"next_cursor"`
}

// Create 新建一条内容（可无文本，由调用方决定是否合法）。
func (s *Store) Create(in CreateInput) (*Item, error) {
	text := strings.TrimSpace(in.Text)
	if s.opts.MaxTextBytes > 0 && len(text) > s.opts.MaxTextBytes {
		return nil, ErrTextTooLong
	}
	it := &Item{
		ID:        NewID(),
		CreatedAt: time.Now().UTC(),
		CreatedBy: in.CreatedBy,
		SourceIP:  in.SourceIP,
		Text:      text,
		Files:     []File{},
	}
	if err := os.MkdirAll(s.filesDir(it.ID), 0o750); err != nil {
		return nil, fmt.Errorf("创建内容目录失败: %w", err)
	}
	if err := s.saveMeta(it); err != nil {
		os.RemoveAll(s.itemDir(it.ID))
		return nil, err
	}
	s.mu.Lock()
	s.index[it.ID] = it
	s.mu.Unlock()
	return it.clone(), nil
}

// Get 返回内容副本。
func (s *Store) Get(id string) (*Item, error) {
	s.mu.RLock()
	defer s.mu.RUnlock()
	it, ok := s.index[id]
	if !ok {
		return nil, ErrNotFound
	}
	return it.clone(), nil
}

// List 按 ID 倒序返回内容，支持游标、搜索与"未取走"过滤。
func (s *Store) List(q ListQuery) (ListResult, error) {
	if q.Limit <= 0 || q.Limit > 200 {
		q.Limit = 50
	}
	needle := strings.ToLower(strings.TrimSpace(q.Query))

	s.mu.RLock()
	ids := make([]string, 0, len(s.index))
	for id := range s.index {
		ids = append(ids, id)
	}
	sort.Sort(sort.Reverse(sort.StringSlice(ids)))
	res := ListResult{Items: []*Item{}}
	for _, id := range ids {
		if q.Before != "" && id >= q.Before {
			continue
		}
		it := s.index[id]
		if q.Todo && it.Taken {
			continue
		}
		if needle != "" && !it.match(needle) {
			continue
		}
		if len(res.Items) >= q.Limit {
			res.NextCursor = res.Items[len(res.Items)-1].ID
			break
		}
		res.Items = append(res.Items, it.clone())
	}
	s.mu.RUnlock()
	return res, nil
}

// SetTaken 标记/取消"已取走"。
func (s *Store) SetTaken(id string, taken bool, by string) (*Item, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	it, ok := s.index[id]
	if !ok {
		return nil, ErrNotFound
	}
	cp := it.clone()
	if taken {
		now := time.Now().UTC()
		cp.Taken = true
		cp.TakenAt = &now
		cp.TakenBy = by
	} else {
		cp.Taken = false
		cp.TakenAt = nil
		cp.TakenBy = ""
	}
	if err := s.saveMeta(cp); err != nil {
		return nil, err
	}
	s.index[id] = cp
	return cp.clone(), nil
}

// Delete 直接删除整条内容（连同目录）。
func (s *Store) Delete(id string) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	if _, ok := s.index[id]; !ok {
		return ErrNotFound
	}
	if err := os.RemoveAll(s.itemDir(id)); err != nil {
		return fmt.Errorf("删除内容失败: %w", err)
	}
	delete(s.index, id)
	return nil
}

import (
	"crypto/sha256"

// AddFile 将 r 流式写入该内容的 files 目录，同时计算 sha256。
// 内存占用恒定，与文件大小无关。
// 调用约定：同一 item 的 AddFile 必须串行调用（httpapi 单次上传内顺序处理），
// 且不存在向已有内容追加附件的 API，因此不会出现并发合并冲突。
func (s *Store) AddFile(itemID, name string, declaredSize int64, r io.Reader) (*Item, error) {
	clean := SanitizeName(name)

	s.mu.RLock()
	cur, ok := s.index[itemID]
	var count int
	if ok {
		count = len(cur.Files)
	}
	s.mu.RUnlock()
	if !ok {
		return nil, ErrNotFound
	}
	if count >= MaxFilesPerItem {
		return nil, ErrTooManyFiles
	}

	if s.opts.FreeSpace != nil {
		free, err := s.opts.FreeSpace(s.itemsRoot)
		if err != nil {
			return nil, fmt.Errorf("检查磁盘空间失败: %w", err)
		}
		need := s.opts.MinFreeBytes
		if declaredSize > 0 {
			need += declaredSize
		}
		if free < need {
			return nil, ErrNoSpace
		}
	}

	var rb [4]byte
	for i := range rb {
		rb[i] = byte(rand.UintN(256))
	}
	fid := "f" + hex.EncodeToString(rb[:])
	diskPath := filepath.Join(s.filesDir(itemID), fid+"__"+clean)

	f, err := os.OpenFile(diskPath, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0o640)
	if err != nil {
		return nil, fmt.Errorf("创建附件文件失败: %w", err)
	}
	hasher := sha256.New()
	max := s.opts.MaxFileBytes
	var src io.Reader = r
	if max > 0 {
		src = io.LimitReader(r, max+1)
	}
	n, copyErr := io.Copy(io.MultiWriter(f, hasher), src)
	closeErr := f.Close()
	if copyErr != nil {
		os.Remove(diskPath)
		return nil, fmt.Errorf("写入附件失败: %w", copyErr)
	}
	if closeErr != nil {
		os.Remove(diskPath)
		return nil, fmt.Errorf("保存附件失败: %w", closeErr)
	}
	if max > 0 && n > max {
		os.Remove(diskPath)
		return nil, ErrTooLarge
	}

	file := File{ID: fid, Name: clean, Size: n, SHA256: hex.EncodeToString(hasher.Sum(nil))}

	s.mu.Lock()
	it, ok := s.index[itemID]
	if !ok {
		s.mu.Unlock()
		os.Remove(diskPath)
		return nil, ErrNotFound
	}
	cp := it.clone()
	cp.Files = append(cp.Files, file)
	if err := s.saveMeta(cp); err != nil {
		s.mu.Unlock()
		os.Remove(diskPath)
		return nil, err
	}
	s.index[itemID] = cp
	s.mu.Unlock()
	return cp.clone(), nil
}

// FilePath 返回附件的磁盘路径与元数据。文件 ID 必须在 meta 中登记，因此不存在路径穿越。
func (s *Store) FilePath(itemID, fileID string) (string, *File, error) {
	if strings.ContainsAny(itemID, `/\`) || itemID == "" {
		return "", nil, ErrNotFound
	}
	s.mu.RLock()
	it, ok := s.index[itemID]
	var found *File
	if ok {
		for i := range it.Files {
			if it.Files[i].ID == fileID {
				f := it.Files[i]
				found = &f
				break
			}
		}
	}
	s.mu.RUnlock()
	if !ok {
		return "", nil, ErrNotFound
	}
	if found == nil {
		return "", nil, ErrFileNotFound
	}
	p := filepath.Join(s.filesDir(itemID), found.ID+"__"+found.Name)
	return p, found, nil
}

// IncDownloads 累加附件下载次数。
func (s *Store) IncDownloads(itemID, fileID string) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	it, ok := s.index[itemID]
	if !ok {
		return ErrNotFound
	}
	cp := it.clone()
	hit := false
	for i := range cp.Files {
		if cp.Files[i].ID == fileID {
			cp.Files[i].Downloads++
			hit = true
			break
		}
	}
	if !hit {
		return ErrFileNotFound
	}
	if err := s.saveMeta(cp); err != nil {
		return err
	}
	s.index[itemID] = cp
	return nil
}

// DeleteFile 删除单个附件。
func (s *Store) DeleteFile(itemID, fileID string) (*Item, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	it, ok := s.index[itemID]
	if !ok {
		return nil, ErrNotFound
	}
	cp := it.clone()
	idx := -1
	for i := range cp.Files {
		if cp.Files[i].ID == fileID {
			idx = i
			break
		}
	}
	if idx < 0 {
		return nil, ErrFileNotFound
	}
	diskPath := filepath.Join(s.filesDir(itemID), cp.Files[idx].ID+"__"+cp.Files[idx].Name)
	cp.Files = append(cp.Files[:idx], cp.Files[idx+1:]...)
	if err := s.saveMeta(cp); err != nil {
		return nil, err
	}
	s.index[itemID] = cp
	if err := os.Remove(diskPath); err != nil && !os.IsNotExist(err) {
		return nil, fmt.Errorf("删除附件文件失败: %w", err)
	}
	return cp.clone(), nil
}

// SanitizeName 清洗用户提交的文件名：仅用于展示与磁盘后缀，绝不参与目录选择。
func SanitizeName(name string) string {
	// 统一分隔符后取最后一段，彻底消除 ../ 之类
	flat := strings.ReplaceAll(name, "\\", "/")
	flat = path.Base(flat)
	var b strings.Builder
	for _, r := range flat {
		if r < 0x20 || r == 0x7f {
			continue
		}
		if strings.ContainsRune(`/\:*?"<>|`, r) {
			continue
		}
		b.WriteRune(r)
	}
	out := strings.TrimSpace(b.String())
	out = strings.TrimLeft(out, ".") // 去掉前导点，避免隐藏文件与 ".."
	out = strings.TrimSpace(out)
	if out == "" || out == "." || out == ".." {
		return "未命名文件"
	}
	const maxRunes = 120
	runes := []rune(out)
	if len(runes) > maxRunes {
		ext := path.Ext(out)
		if len([]rune(ext)) > 20 || ext == out {
			ext = ""
		}
		keep := maxRunes - len([]rune(ext))
		out = string(runes[:keep]) + ext
	}
	return out
}

)

// PBKDF2Iter 是哈希迭代次数（OWASP 2023 对 PBKDF2-HMAC-SHA256 的建议量级）。
const PBKDF2Iter = 600_000

const (
	saltLen = 16
	keyLen  = 32
	// MinPasswordLen 是密码最小长度。
	MinPasswordLen = 8
)

// ErrWeakPassword 表示密码过短。
var ErrWeakPassword = fmt.Errorf("密码至少需要 %d 位", MinPasswordLen)

// HashPassword 返回 base64 盐、base64 派生密钥与迭代次数。
func HashPassword(password string) (salt, hash string, iter int, err error) {
	raw := make([]byte, saltLen)
	if _, err := rand.Read(raw); err != nil {
		return "", "", 0, fmt.Errorf("生成盐失败: %w", err)
	}
	key, err := pbkdf2.Key(sha256.New, password, raw, PBKDF2Iter, keyLen)
	if err != nil {
		return "", "", 0, fmt.Errorf("派生密钥失败: %w", err)
	}
	return base64.StdEncoding.EncodeToString(raw),
		base64.StdEncoding.EncodeToString(key),
		PBKDF2Iter, nil
}

// VerifyPassword 用常量时间比较校验密码。
func VerifyPassword(password, saltB64, hashB64 string, iter int) bool {
	salt, err := base64.StdEncoding.DecodeString(saltB64)
	if err != nil {
		return false
	}
	want, err := base64.StdEncoding.DecodeString(hashB64)
	if err != nil || len(want) == 0 {
		return false
	}
	if iter <= 0 {
		iter = PBKDF2Iter
	}
	got, err := pbkdf2.Key(sha256.New, password, salt, iter, len(want))
	if err != nil {
		return false
	}
	return subtle.ConstantTimeCompare(got, want) == 1
}

const passwordAlphabet = "abcdefghijkmnpqrstuvwxyzABCDEFGHJKLMNPQRSTUVWXYZ23456789"

// GeneratePassword 生成 16 位随机密码，排除易混字符（0/O/1/l/I）。
func GeneratePassword() string {
	const n = 16
	out := make([]byte, n)
	max := big.NewInt(int64(len(passwordAlphabet)))
	for i := 0; i < n; i++ {
		idx, err := rand.Int(rand.Reader, max)
		if err != nil {
			panic(err) // crypto/rand 失败意味着系统熵不可用
		}
		out[i] = passwordAlphabet[idx.Int64()]
	}
	return string(out)
}

func validatePassword(pw string) error {
	if len(pw) < MinPasswordLen {
		return ErrWeakPassword
	}
	return nil
}

package auth

import (
	"crypto/rand"
	"encoding/base64"
	"encoding/hex"
	"encoding/json"
	"errors"
	"fmt"
	"os"
	"path/filepath"
	"sort"
	"sync"
	"time"
)

var (
	// ErrUserExists 表示用户名已存在。
	ErrUserExists = errors.New("用户名已存在")
	// ErrUserNotFound 表示用户名不存在。
	ErrUserNotFound = errors.New("用户不存在")
	// ErrSelfDelete 表示试图删除自己。
	ErrSelfDelete = errors.New("不能删除当前登录的账号")
	// ErrLastAdmin 表示试图删除最后一个管理员。
	ErrLastAdmin = errors.New("不能删除最后一个管理员")
	// ErrBadCredentials 表示用户名或密码错误。
	ErrBadCredentials = errors.New("用户名或密码错误")
)

// User 是一个账号。Salt/Hash/Iter 用于密码校验。
type User struct {
	Username  string    `json:"username"`
	Salt      string    `json:"salt"`
	Hash      string    `json:"hash"`
	Iter      int       `json:"iter"`
	IsAdmin   bool      `json:"is_admin"`
	CreatedAt time.Time `json:"created_at"`
}

// Session 是一个内存会话。
type Session struct {
	Username string
	Expires  time.Time
}

// Manager 管理用户库（持久化到 path）与会话（仅内存）。
type Manager struct {
	path     string
	ttl      time.Duration
	mu       sync.RWMutex
	users    map[string]*User
	sessions map[string]*Session
	limiter  *Limiter
	now      func() time.Time
}

type userFile struct {
	Users []*User `json:"users"`
}

// Open 打开（或初始化）用户库。
func Open(path string, ttl time.Duration) (*Manager, error) {
	m := &Manager{
		path:     path,
		ttl:      ttl,
		users:    map[string]*User{},
		sessions: map[string]*Session{},
		limiter:  NewLimiter(15*time.Minute, 5, 10*time.Minute),
		now:      time.Now,
	}
	m.limiter.now = m.now
	if err := os.MkdirAll(filepath.Dir(path), 0o750); err != nil {
		return nil, fmt.Errorf("创建用户库目录失败: %w", err)
	}
	raw, err := os.ReadFile(path)
	switch {
	case err == nil:
		var f userFile
		if err := json.Unmarshal(raw, &f); err != nil {
			return nil, fmt.Errorf("解析用户库 %s 失败: %w", path, err)
		}
		for _, u := range f.Users {
			if u != nil && u.Username != "" {
				m.users[u.Username] = u
			}
		}
	case errors.Is(err, os.ErrNotExist):
	default:
		return nil, fmt.Errorf("读取用户库失败: %w", err)
	}
	return m, nil
}

// BootstrapAdmin 在用户库为空时创建 admin 并返回随机密码。
func (m *Manager) BootstrapAdmin() (password string, created bool, err error) {
	m.mu.RLock()
	empty := len(m.users) == 0
	m.mu.RUnlock()
	if !empty {
		return "", false, nil
	}
	pw := GeneratePassword()
	if err := m.CreateUser("admin", pw, true); err != nil {
		return "", false, err
	}
	return pw, true, nil
}

// Verify 校验用户名密码。
func (m *Manager) Verify(username, password string) (*User, bool) {
	m.mu.RLock()
	u, ok := m.users[username]
	var salt, hash string
	var iter int
	if ok {
		salt, hash, iter = u.Salt, u.Hash, u.Iter
	}
	m.mu.RUnlock()
	if !ok {
		// 用户不存在也做一次哈希，避免时序泄漏用户是否存在
		VerifyPassword(password, "AAAAAAAAAAAAAAAAAAAAAA==", "AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA=", PBKDF2Iter)
		return nil, false
	}
	if !VerifyPassword(password, salt, hash, iter) {
		return nil, false
	}
	cp := *u
	return &cp, true
}

// CreateUser 新建账号。
func (m *Manager) CreateUser(username, password string, isAdmin bool) error {
	if err := validatePassword(password); err != nil {
		return err
	}
	if username == "" {
		return errors.New("用户名不能为空")
	}
	salt, hash, iter, err := HashPassword(password)
	if err != nil {
		return err
	}
	m.mu.Lock()
	defer m.mu.Unlock()
	if _, ok := m.users[username]; ok {
		return ErrUserExists
	}
	m.users[username] = &User{
		Username: username, Salt: salt, Hash: hash, Iter: iter,
		IsAdmin: isAdmin, CreatedAt: m.now().UTC(),
	}
	return m.saveLocked()
}

// SetPassword 重设密码。
func (m *Manager) SetPassword(username, password string) error {
	if err := validatePassword(password); err != nil {
		return err
	}
	salt, hash, iter, err := HashPassword(password)
	if err != nil {
		return err
	}
	m.mu.Lock()
	defer m.mu.Unlock()
	u, ok := m.users[username]
	if !ok {
		return ErrUserNotFound
	}
	u.Salt, u.Hash, u.Iter = salt, hash, iter
	// 改密后踢掉该用户所有会话
	for t, s := range m.sessions {
		if s.Username == username {
			delete(m.sessions, t)
		}
	}
	return m.saveLocked()
}

// DeleteUser 删除账号，actor 是当前操作者。
func (m *Manager) DeleteUser(actor, target string) error {
	m.mu.Lock()
	defer m.mu.Unlock()
	if _, ok := m.users[target]; !ok {
		return ErrUserNotFound
	}
	if actor == target {
		return ErrSelfDelete
	}
	if m.users[target].IsAdmin && m.adminCountLocked() <= 1 {
		return ErrLastAdmin
	}
	delete(m.users, target)
	for t, s := range m.sessions {
		if s.Username == target {
			delete(m.sessions, t)
		}
	}
	return m.saveLocked()
}

func (m *Manager) adminCountLocked() int {
	n := 0
	for _, u := range m.users {
		if u.IsAdmin {
			n++
		}
	}
	return n
}

// List 返回按用户名排序的用户（不含哈希）。
func (m *Manager) List() []*User {
	m.mu.RLock()
	defer m.mu.RUnlock()
	out := make([]*User, 0, len(m.users))
	for _, u := range m.users {
		cp := *u
		cp.Salt, cp.Hash, cp.Iter = "", "", 0
		out = append(out, &cp)
	}
	sort.Slice(out, func(i, j int) bool { return out[i].Username < out[j].Username })
	return out
}

// NewSession 为已校验的用户创建会话令牌。
func (m *Manager) NewSession(username string) (token string, expires time.Time, err error) {
	m.mu.Lock()
	defer m.mu.Unlock()
	if _, ok := m.users[username]; !ok {
		return "", time.Time{}, ErrUserNotFound
	}
	var raw [32]byte
	if _, err := rand.Read(raw[:]); err != nil {
		return "", time.Time{}, fmt.Errorf("生成会话令牌失败: %w", err)
	}
	token = hex.EncodeToString(raw[:])
	expires = m.now().Add(m.ttl)
	m.sessions[token] = &Session{Username: username, Expires: expires}
	return token, expires, nil
}

// Lookup 校验会话并做滑动续期。
func (m *Manager) Lookup(token string) (*User, bool) {
	if token == "" {
		return nil, false
	}
	m.mu.Lock()
	defer m.mu.Unlock()
	s, ok := m.sessions[token]
	if !ok {
		return nil, false
	}
	now := m.now()
	if now.After(s.Expires) {
		delete(m.sessions, token)
		return nil, false
	}
	u, ok := m.users[s.Username]
	if !ok {
		delete(m.sessions, token)
		return nil, false
	}
	s.Expires = now.Add(m.ttl)
	cp := *u
	cp.Salt, cp.Hash, cp.Iter = "", "", 0
	return &cp, true
}

// Drop 删除单个会话（登出）。
func (m *Manager) Drop(token string) {
	m.mu.Lock()
	delete(m.sessions, token)
	m.mu.Unlock()
}

// Limiter 返回登录失败限速器。
func (m *Manager) Limiter() *Limiter { return m.limiter }

// saveLocked 原子写用户库。调用方必须持有写锁。
func (m *Manager) saveLocked() error {
	f := userFile{Users: make([]*User, 0, len(m.users))}
	for _, u := range m.users {
		f.Users = append(f.Users, u)
	}
	sort.Slice(f.Users, func(i, j int) bool { return f.Users[i].Username < f.Users[j].Username })
	raw, err := json.MarshalIndent(f, "", "  ")
	if err != nil {
		return fmt.Errorf("序列化用户库失败: %w", err)
	}
	tmp := m.path + ".tmp"
	fh, err := os.OpenFile(tmp, os.O_WRONLY|os.O_CREATE|os.O_TRUNC, 0o600)
	if err != nil {
		return fmt.Errorf("写入用户库失败: %w", err)
	}
	if _, err := fh.Write(raw); err != nil {
		fh.Close()
		os.Remove(tmp)
		return fmt.Errorf("写入用户库失败: %w", err)
	}
	if err := fh.Sync(); err != nil {
		fh.Close()
		os.Remove(tmp)
		return fmt.Errorf("同步用户库失败: %w", err)
	}
	if err := fh.Close(); err != nil {
		os.Remove(tmp)
		return fmt.Errorf("关闭用户库失败: %w", err)
	}
	if err := os.Rename(tmp, m.path); err != nil {
		os.Remove(tmp)
		return fmt.Errorf("替换用户库失败: %w", err)
	}
	return nil
}

```go
package auth

import (
	"sync"
	"time"
)

type limitEntry struct {
	fails       int
	windowStart time.Time
	lockedUntil time.Time
}

// Limiter 按来源 IP 限制登录失败次数。
type Limiter struct {
	window  time.Duration
	maxFails int
	lock    time.Duration
	now     func() time.Time
	mu      sync.Mutex
	entries map[string]*limitEntry
}

// NewLimiter 创建限速器：window 内失败 maxFails 次则锁 lock 时长。
func NewLimiter(window time.Duration, maxFails int, lock time.Duration) *Limiter {
	return &Limiter{
		window: window, maxFails: maxFails, lock: lock,
		now:     time.Now,
		entries: map[string]*limitEntry{},
	}
}

// Allowed 报告该 IP 是否允许尝试登录。
func (l *Limiter) Allowed(ip string) bool {
	if l == nil {
		return true
	}
	l.mu.Lock()
	defer l.mu.Unlock()
	e, ok := l.entries[ip]
	if !ok {
		return true
	}
	now := l.now()
	if now.Before(e.lockedUntil) {
		return false
	}
	if l.lock > 0 && !e.lockedUntil.IsZero() && now.After(e.lockedUntil) {
		delete(l.entries, ip)
	}
	return true
}

// Fail 记录一次失败；达到阈值时开始锁定。
func (l *Limiter) Fail(ip string) {
	if l == nil {
		return
	}
	l.mu.Lock()
	defer l.mu.Unlock()
	now := l.now()
	e, ok := l.entries[ip]
	if !ok || now.Sub(e.windowStart) > l.window {
		l.entries[ip] = &limitEntry{fails: 1, windowStart: now}
		return
	}
	e.fails++
	if e.fails >= l.maxFails {
		e.lockedUntil = now.Add(l.lock)
	}
}

// Reset 清空该 IP 的失败记录（登录成功时调用）。
func (l *Limiter) Reset(ip string) {
	if l == nil {
		return
	}
	l.mu.Lock()
	delete(l.entries, ip)
	l.mu.Unlock()
}

// errorBody 是统一的错误响应体。
type errorBody struct {
	Error struct {
		Code    string `json:"code"`
		Message string `json:"message"`
	} `json:"error"`
}

// writeJSON 写出 JSON 响应。
func writeJSON(w http.ResponseWriter, status int, v any) {
	w.Header().Set("Content-Type", "application/json; charset=utf-8")
	w.WriteHeader(status)
	if v == nil {
		return
	}
	if err := json.NewEncoder(w).Encode(v); err != nil {
		// 响应已开始写出，只能记录（日志由中间件兜底）
		return
	}
}

// writeError 写出统一错误响应。
func writeError(w http.ResponseWriter, status int, code, msg string) {
	var eb errorBody
	eb.Error.Code = code
	eb.Error.Message = msg
	writeJSON(w, status, eb)
}

// writeStoreErr 把 store 的领域错误映射为状态码与中文消息。
func (s *Server) writeStoreErr(w http.ResponseWriter, err error) {
	switch {
	case errors.Is(err, store.ErrNotFound):
		writeError(w, http.StatusNotFound, "not_found", "内容不存在")
	case errors.Is(err, store.ErrFileNotFound):
		writeError(w, http.StatusNotFound, "file_not_found", "文件不存在")
	case errors.Is(err, store.ErrTooLarge):
		writeError(w, http.StatusRequestEntityTooLarge, "file_too_large",
			fmt.Sprintf("文件超过 %dMB 上限", s.cfg.MaxFileSizeMB))
	case errors.Is(err, store.ErrNoSpace):
		writeError(w, http.StatusInsufficientStorage, "no_space",
			fmt.Sprintf("磁盘空间不足，需保留至少 %dMB 空闲", s.cfg.MinFreeDiskMB))
	case errors.Is(err, store.ErrTooManyFiles):
		writeError(w, http.StatusBadRequest, "too_many_files",
			fmt.Sprintf("附件数量不能超过 %d 个", store.MaxFilesPerItem))
	case errors.Is(err, store.ErrTextTooLong):
		writeError(w, http.StatusBadRequest, "text_too_long",
			fmt.Sprintf("文本超过 %d 字节上限", s.cfg.MaxTextBytes))
	default:
		s.log.Printf("error: %v", err)
		writeError(w, http.StatusInternalServerError, "internal", "服务器内部错误")
	}
}

// clientIP 返回直连客户端 IP（不信任 X-Forwarded-For，避免伪造来源标注）。
func clientIP(r *http.Request) string {
	host, _, err := net.SplitHostPort(r.RemoteAddr)
	if err != nil {
		return r.RemoteAddr
	}
	return host
}

// decodeJSON 读取并解析有大小上限的 JSON 请求体。
func decodeJSON(w http.ResponseWriter, r *http.Request, dst any) bool {
	limited := http.MaxBytesReader(w, r.Body, 1<<20)
	dec := json.NewDecoder(limited)
	dec.DisallowUnknownFields()
	if err := dec.Decode(dst); err != nil {
		writeError(w, http.StatusBadRequest, "bad_request", "请求内容格式不正确")
		return false
	}
	return true
}

// attachmentHeader 构造同时兼容老浏览器与 UTF-8 文件名的 Content-Disposition。
func attachmentHeader(name string) string {
	ascii := strings.Map(func(r rune) rune {
		if r < 0x20 || r > 0x7e || r == '"' || r == '\\' {
			return '_'
		}
		return r
	}, name)
	return fmt.Sprintf("attachment; filename=%q; filename*=UTF-8''%s", ascii, rfc5987(name))
}

// rfc5987 按 RFC 5987 对文件名做百分号编码。
func rfc5987(s string) string {
	var b strings.Builder
	for _, c := range []byte(s) {
		switch {
		case c >= 'a' && c <= 'z', c >= 'A' && c <= 'Z', c >= '0' && c <= '9',
			c == '-', c == '.', c == '_', c == '~':
			b.WriteByte(c)
		default:
			fmt.Fprintf(&b, "%%%02X", c)
		}
	}
	return b.String()
}

import (

type ctxKey int

const (
	ctxUser ctxKey = iota
	ctxToken

// chain 组合所有全局中间件。
func (s *Server) chain(next http.Handler) http.Handler {
	return s.recoverMW(s.securityHeaders(s.accessLog(s.csrf(next))))
}

func (s *Server) recoverMW(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		defer func() {
			if rec := recover(); rec != nil {
				s.log.Printf("panic: %v\n%s", rec, debug.Stack())
				writeError(w, http.StatusInternalServerError, "panic", "服务器内部错误")
			}
		}()
		next.ServeHTTP(w, r)
	})
}

func (s *Server) securityHeaders(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		h := w.Header()
		h.Set("Content-Security-Policy", "default-src 'self'")
		h.Set("X-Content-Type-Options", "nosniff")
		h.Set("X-Frame-Options", "DENY")
		h.Set("Referrer-Policy", "no-referrer")
		next.ServeHTTP(w, r)
	})
}

type logWriter struct {
	http.ResponseWriter
	status int
	size   int64
}

func (l *logWriter) WriteHeader(code int) {
	if l.status == 0 {
		l.status = code
	}
	l.ResponseWriter.WriteHeader(code)
}

func (l *logWriter) Write(b []byte) (int, error) {
	if l.status == 0 {
		l.status = http.StatusOK
	}
	n, err := l.ResponseWriter.Write(b)
	l.size += int64(n)
	return n, err
}

// Flush 让 SSE 能即时下发（包装后仍满足 http.Flusher）。
func (l *logWriter) Flush() {
	if f, ok := l.ResponseWriter.(http.Flusher); ok {
		f.Flush()
	}
}

func (s *Server) accessLog(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		start := time.Now()
		lw := &logWriter{ResponseWriter: w}
		next.ServeHTTP(lw, r)
		if strings.HasPrefix(r.URL.Path, "/api/") {
			s.log.Printf("%s %s %d %dB %s %s", r.Method, r.URL.Path, lw.status, lw.size,
				clientIP(r), time.Since(start).Round(time.Millisecond))
		}
	})
}

// csrf 校验不安全方法的 Origin/Referer 必须与 Host 同源。
func (s *Server) csrf(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		switch r.Method {
		case http.MethodGet, http.MethodHead, http.MethodOptions:
			next.ServeHTTP(w, r)
			return
		}
		if origin := r.Header.Get("Origin"); origin != "" {
			if !sameHost(origin, r.Host) {
				writeError(w, http.StatusForbidden, "csrf", "请求来源不被信任")
				return
			}
		} else if ref := r.Header.Get("Referer"); ref != "" {
			if !sameHost(ref, r.Host) {
				writeError(w, http.StatusForbidden, "csrf", "请求来源不被信任")
				return
			}
		}
		next.ServeHTTP(w, r)
	})
}

// sameHost 比较 URL 的 Host 与请求 Host。
func sameHost(rawURL, host string) bool {
	u, err := url.Parse(rawURL)
	if err != nil {
		return false
	}
	return strings.EqualFold(u.Host, host)
}

// withAuth 校验会话，把用户放进 context，并做滑动续期。
func (s *Server) withAuth(h http.HandlerFunc) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		c, err := r.Cookie(sessionCookie)
		if err != nil || c.Value == "" {
			writeError(w, http.StatusUnauthorized, "unauthorized", "请先登录")
			return
		}
		u, ok := s.auth.Lookup(c.Value)
		if !ok {
			writeError(w, http.StatusUnauthorized, "unauthorized", "会话已过期，请重新登录")
			return
		}
		http.SetCookie(w, &http.Cookie{
			Name: sessionCookie, Value: c.Value, Path: "/",
			HttpOnly: true, Secure: true, SameSite: http.SameSiteLaxMode,
			Expires: time.Now().Add(s.cfg.SessionTTL()),
		})
		ctx := context.WithValue(r.Context(), ctxUser, u)
		ctx = context.WithValue(ctx, ctxToken, c.Value)
		h(w, r.WithContext(ctx))
	}
}

// withAdmin 在 withAuth 基础上要求管理员。
func (s *Server) withAdmin(h http.HandlerFunc) http.HandlerFunc {
	return s.withAuth(func(w http.ResponseWriter, r *http.Request) {
		u := userFrom(r)
		if u == nil || !u.IsAdmin {
			writeError(w, http.StatusForbidden, "forbidden", "需要管理员权限")
			return
		}
		h(w, r)
	})
}

// userFrom 取出当前登录用户（已由 withAuth 注入）。
func userFrom(r *http.Request) *auth.User {
	u, _ := r.Context().Value(ctxUser).(*auth.User)
	return u
}

func tokenFrom(r *http.Request) string {
	t, _ := r.Context().Value(ctxToken).(string)
	return t
}

type loginInput struct {
	Username string `json:"username"`
	Password string `json:"password"`
}

type meOutput struct {
	Username string `json:"username"`
	IsAdmin  bool   `json:"is_admin"`
}

func (s *Server) handleLogin(w http.ResponseWriter, r *http.Request) {
	var in loginInput
	if !decodeJSON(w, r, &in) {
		return
	}
	ip := clientIP(r)
	if !s.auth.Limiter().Allowed(ip) {
		writeError(w, http.StatusForbidden, "login_locked", "登录失败次数过多，请稍后再试")
		return
	}
	u, ok := s.auth.Verify(in.Username, in.Password)
	if !ok {
		s.auth.Limiter().Fail(ip)
		writeError(w, http.StatusUnauthorized, "bad_credentials", "用户名或密码错误")
		return
	}
	s.auth.Limiter().Reset(ip)
	token, expires, err := s.auth.NewSession(u.Username)
	if err != nil {
		s.writeStoreErr(w, err)
		return
	}
	http.SetCookie(w, &http.Cookie{
		Name: sessionCookie, Value: token, Path: "/",
		HttpOnly: true, Secure: true, SameSite: http.SameSiteLaxMode, Expires: expires,
	})
	s.log.Printf("info: %s 登录成功", u.Username)
	writeJSON(w, http.StatusOK, meOutput{Username: u.Username, IsAdmin: u.IsAdmin})
}

func (s *Server) handleLogout(w http.ResponseWriter, r *http.Request) {
	s.auth.Drop(tokenFrom(r))
	http.SetCookie(w, &http.Cookie{
		Name: sessionCookie, Value: "", Path: "/",
		HttpOnly: true, Secure: true, SameSite: http.SameSiteLaxMode,
		Expires: time.Unix(0, 0),
	})
	w.WriteHeader(http.StatusNoContent)
}

func (s *Server) handleMe(w http.ResponseWriter, r *http.Request) {
	u := userFrom(r)
	writeJSON(w, http.StatusOK, meOutput{Username: u.Username, IsAdmin: u.IsAdmin})
}

func (s *Server) handleChangePassword(w http.ResponseWriter, r *http.Request) {
	var in struct {
		Old string `json:"old"`
		New string `json:"new"`
	}
	if !decodeJSON(w, r, &in) {
		return
	}
	u := userFrom(r)
	if _, ok := s.auth.Verify(u.Username, in.Old); !ok {
		writeError(w, http.StatusForbidden, "bad_credentials", "当前密码不正确")
		return
	}
	if err := s.auth.SetPassword(u.Username, in.New); err != nil {
		writeError(w, http.StatusBadRequest, "weak_password", err.Error())
		return
	}
	// 改密会踢掉该用户所有会话，这里顺手清掉 Cookie
	http.SetCookie(w, &http.Cookie{
		Name: sessionCookie, Value: "", Path: "/",
		HttpOnly: true, Secure: true, SameSite: http.SameSiteLaxMode,
		Expires: time.Unix(0, 0),
	})
	w.WriteHeader(http.StatusNoContent)
}

`internal/httpapi/sse.go`：
```go
package httpapi

import "sync"

// Event 是一条 SSE 事件。Data 已是 JSON 文本。
type Event struct {
	Type string
	Data string
}

// Hub 维护 SSE 订阅者集合。
type Hub struct {
	mu   sync.Mutex
	subs map[chan Event]struct{}
}

// NewHub 创建空集线器。
func NewHub() *Hub {
	return &Hub{subs: map[chan Event]struct{}{}}
}

// Subscribe 返回事件通道与取消函数。
func (h *Hub) Subscribe() (<-chan Event, func()) {
	ch := make(chan Event, 16)
	h.mu.Lock()
	h.subs[ch] = struct{}{}
	h.mu.Unlock()
	cancel := func() {
		h.mu.Lock()
		if _, ok := h.subs[ch]; ok {
			delete(h.subs, ch)
			close(ch)
		}
		h.mu.Unlock()
	}
	return ch, cancel
}

// Broadcast 非阻塞广播；消费过慢的订阅者会被断开。
func (h *Hub) Broadcast(e Event) {
	h.mu.Lock()
	defer h.mu.Unlock()
	for ch := range h.subs {
		select {
		case ch <- e:
		default:
			delete(h.subs, ch)
			close(ch)
		}
	}
}

// Count 返回当前订阅者数量。
func (h *Hub) Count() int {
	h.mu.Lock()
	defer h.mu.Unlock()
	return len(h.subs)
}

// Close 关闭全部订阅者（优雅退出用）。
func (h *Hub) Close() {
	h.mu.Lock()
	defer h.mu.Unlock()
	for ch := range h.subs {
		delete(h.subs, ch)
		close(ch)
	}
}

	cp.Text = text

// handleCreateItem 以 multipart 流式接收文本与附件。
// 附件可能先于文本到达，因此先建内容（空文本），收完再回填文本。
func (s *Server) handleCreateItem(w http.ResponseWriter, r *http.Request) {
	user := userFrom(r)
	mr, err := r.MultipartReader()
	if err != nil {
		writeError(w, http.StatusBadRequest, "bad_request", "请求必须是 multipart/form-data")
		return
	}

	var (
		text  string
		item  *store.Item
		ip    = clientIP(r)
		files int
	)
	cleanup := func() {
		if item != nil {
			if err := s.store.Delete(item.ID); err != nil {
				s.log.Printf("warn: 清理失败的上传 %s 失败: %v", item.ID, err)
			}
			item = nil
		}
	}

	for {
		part, err := mr.NextPart()
		if errors.Is(err, io.EOF) {
			break
		}
		if err != nil {
			cleanup()
			writeError(w, http.StatusBadRequest, "bad_request", "读取上传数据失败: "+err.Error())
			return
		}
		if part.FormName() == "text" {
			raw, err := io.ReadAll(io.LimitReader(part, int64(s.cfg.MaxTextBytes)+1))
			part.Close()
			if err != nil {
				cleanup()
				writeError(w, http.StatusBadRequest, "bad_request", "读取文本失败")
				return
			}
			if len(raw) > s.cfg.MaxTextBytes {
				cleanup()
				writeError(w, http.StatusBadRequest, "text_too_long",
					"文本超过 "+strconv.Itoa(s.cfg.MaxTextBytes)+" 字节上限")
				return
			}
			text = string(raw)
			continue
		}
		name := part.FileName()
		if name == "" {
			part.Close()
			continue // 非文件字段，忽略
		}
		if item == nil {
			created, err := s.store.Create(store.CreateInput{CreatedBy: user.Username, SourceIP: ip})
			if err != nil {
				part.Close()
				s.writeStoreErr(w, err)
				return
			}
			item = created
		}
		updated, err := s.store.AddFile(item.ID, name, -1, part)
		part.Close()
		if err != nil {
			cleanup()
			s.writeStoreErr(w, err)
			return
		}
		item = updated
		files++
	}

	if item == nil {
		if strings.TrimSpace(text) == "" {
			writeError(w, http.StatusBadRequest, "empty", "请输入文字或选择文件")
			return
		}
		created, err := s.store.Create(store.CreateInput{Text: text, CreatedBy: user.Username, SourceIP: ip})
		if err != nil {
			s.writeStoreErr(w, err)
			return
		}
		item = created
	} else if strings.TrimSpace(text) != "" {
		updated, err := s.store.SetText(item.ID, text)
		if err != nil {
			cleanup()
			s.writeStoreErr(w, err)
			return
		}
		item = updated
	}

	s.log.Printf("info: %s 从 %s 发布内容 %s（%d 个附件）", user.Username, ip, item.ID, files)
	s.broadcast("item.created", map[string]any{"item": item})
	writeJSON(w, http.StatusCreated, item)
}

// handleDownload 流式下发附件，支持 Range 断点续传。
func (s *Server) handleDownload(w http.ResponseWriter, r *http.Request) {
	id, fid := r.PathValue("id"), r.PathValue("fid")
	path, f, err := s.store.FilePath(id, fid)
	if err != nil {
		s.writeStoreErr(w, err)
		return
	}
	fh, err := os.Open(path)
	if err != nil {
		s.log.Printf("warn: 打开附件 %s 失败: %v", path, err)
		writeError(w, http.StatusNotFound, "file_missing", "文件已不存在")
		return
	}
	defer fh.Close()
	st, err := fh.Stat()
	if err != nil {
		s.writeStoreErr(w, err)
		return
	}

	h := w.Header()
	h.Set("Content-Type", "application/octet-stream")
	h.Set("Content-Disposition", attachmentHeader(f.Name))
	h.Set("Content-Length", strconv.FormatInt(st.Size(), 10))
	h.Set("Cache-Control", "no-store")

	http.ServeContent(w, r, "", st.ModTime(), fh)

	if err := s.store.IncDownloads(id, fid); err != nil {
		s.log.Printf("warn: 下载计数失败 %s/%s: %v", id, fid, err)
	}
}

// handleEvents 建立 SSE 长连接：先握手，再持续下发事件与心跳。
func (s *Server) handleEvents(w http.ResponseWriter, r *http.Request) {
	flusher, ok := w.(http.Flusher)
	if !ok {
		writeError(w, http.StatusInternalServerError, "no_sse", "当前连接不支持 SSE")
		return
	}
	h := w.Header()
	h.Set("Content-Type", "text/event-stream; charset=utf-8")
	h.Set("Cache-Control", "no-cache")
	h.Set("X-Accel-Buffering", "no")
	w.WriteHeader(http.StatusOK)
	fmt.Fprint(w, ": connected\n\n")
	flusher.Flush()

	ch, cancel := s.hub.Subscribe()
	defer cancel()

	ticker := time.NewTicker(25 * time.Second)
	defer ticker.Stop()

	for {
		select {
		case <-r.Context().Done():
			return
		case ev, open := <-ch:
			if !open {
				return
			}
			if _, err := fmt.Fprintf(w, "event: %s\ndata: %s\n\n", ev.Type, ev.Data); err != nil {
				return
			}
			flusher.Flush()
		case <-ticker.C:
			if _, err := fmt.Fprint(w, ": heartbeat\n\n"); err != nil {
				return
			}
			flusher.Flush()
		}
	}
}

package httpapi

import (
	"errors"
	"net/http"

	"lanbox/internal/auth"

// userDTO 是对外暴露的用户信息（不含任何哈希材料）。
type userDTO struct {
	Username  string `json:"username"`
	IsAdmin   bool   `json:"is_admin"`
	CreatedAt string `json:"created_at"`
}

func toDTO(users []*auth.User) []userDTO {
	out := make([]userDTO, 0, len(users))
	for _, u := range users {
		out = append(out, userDTO{
			Username:  u.Username,
			IsAdmin:   u.IsAdmin,
			CreatedAt: u.CreatedAt.UTC().Format("2006-01-02T15:04:05Z"),
		})
	}
	return out
}

func (s *Server) handleListUsers(w http.ResponseWriter, r *http.Request) {
	writeJSON(w, http.StatusOK, toDTO(s.auth.List()))
}

func (s *Server) handleCreateUser(w http.ResponseWriter, r *http.Request) {
	var in struct {
		Username string `json:"username"`
		Password string `json:"password"`
		IsAdmin  bool   `json:"is_admin"`
	}
	if !decodeJSON(w, r, &in) {
		return
	}
	if err := s.auth.CreateUser(in.Username, in.Password, in.IsAdmin); err != nil {
		switch {
		case errors.Is(err, auth.ErrUserExists):
			writeError(w, http.StatusConflict, "user_exists", "用户名已存在")
		case errors.Is(err, auth.ErrWeakPassword):
			writeError(w, http.StatusBadRequest, "weak_password", err.Error())
		default:
			writeError(w, http.StatusBadRequest, "bad_request", err.Error())
		}
		return
	}
	s.log.Printf("info: %s 新增用户 %s（管理员=%v）", userFrom(r).Username, in.Username, in.IsAdmin)
	writeJSON(w, http.StatusCreated, userDTO{Username: in.Username, IsAdmin: in.IsAdmin})
}

func (s *Server) handleSetUserPassword(w http.ResponseWriter, r *http.Request) {
	var in struct {
		Password string `json:"password"`
	}
	if !decodeJSON(w, r, &in) {
		return
	}
	name := r.PathValue("name")
	if err := s.auth.SetPassword(name, in.Password); err != nil {
		switch {
		case errors.Is(err, auth.ErrUserNotFound):
			writeError(w, http.StatusNotFound, "user_not_found", "用户不存在")
		case errors.Is(err, auth.ErrWeakPassword):
			writeError(w, http.StatusBadRequest, "weak_password", err.Error())
		default:
			writeError(w, http.StatusInternalServerError, "internal", "服务器内部错误")
		}
		return
	}
	s.log.Printf("info: %s 重置了用户 %s 的密码", userFrom(r).Username, name)
	w.WriteHeader(http.StatusNoContent)
}

func (s *Server) handleDeleteUser(w http.ResponseWriter, r *http.Request) {
	name := r.PathValue("name")
	err := s.auth.DeleteUser(userFrom(r).Username, name)
	switch {
	case err == nil:
		s.log.Printf("info: %s 删除了用户 %s", userFrom(r).Username, name)
		w.WriteHeader(http.StatusNoContent)
	case errors.Is(err, auth.ErrUserNotFound):
		writeError(w, http.StatusNotFound, "user_not_found", "用户不存在")
	case errors.Is(err, auth.ErrSelfDelete):
		writeError(w, http.StatusForbidden, "self_delete", "不能删除当前登录的账号")
	case errors.Is(err, auth.ErrLastAdmin):
		writeError(w, http.StatusForbidden, "last_admin", "不能删除最后一个管理员")
	default:
		writeError(w, http.StatusInternalServerError, "internal", "服务器内部错误")
	}
}

package main

import (
	"context"
	"crypto/tls"
	"flag"
	"fmt"
	"log"
	"net/http"

	"lanbox/internal/tlsutil"
)

func main() {
	if err := run(); err != nil {
		log.Fatalf("lanbox 启动失败: %v", err)
	}
}

func run() error {
	cfgPath := flag.String("config", "config.json", "配置文件路径（不存在则使用默认值）")
	resetAdmin := flag.Bool("reset-admin-password", false, "重置 admin 密码并退出")
	flag.Parse()

	logger := log.New(os.Stdout, "", log.LstdFlags)
	cfg, err := config.Load(*cfgPath)
	if err != nil {
		return err
	}
	if err := os.MkdirAll(cfg.DataDir, 0o750); err != nil {
		return fmt.Errorf("创建数据目录 %s 失败: %w", cfg.DataDir, err)
	}

	am, err := auth.Open(filepath.Join(cfg.DataDir, "users.json"), cfg.SessionTTL())
	if err != nil {
		return err
	}

	if *resetAdmin {
		pw := auth.GeneratePassword()
		if err := am.SetPassword("admin", pw); err != nil {
			return err
		}
		printPasswordBox("admin 密码已重置", pw)
		return nil
	}

	pw, created, err := am.BootstrapAdmin()
	if err != nil {
		return err
	}
	if created {
		printPasswordBox("已创建管理员账号 admin", pw)
	}

	st, err := store.Open(filepath.Join(cfg.DataDir, "items"), store.Options{
		MaxFileBytes: cfg.MaxFileBytes(),
		MinFreeBytes: cfg.MinFreeBytes(),
	})
	if err != nil {
		return err
	}

	certPath, keyPath, generated, err := tlsutil.Ensure(cfg.DataDir, cfg.TLS.CertFile, cfg.TLS.KeyFile, tlsutil.Hosts(cfg.Listen))
	if err != nil {
		return err
	}
	if generated {
		logger.Printf("info: 已生成自签证书 %s（浏览器首次访问需手动信任）", certPath)
	}

	srv := httpapi.New(httpapi.Options{Cfg: cfg, Store: st, Auth: am, Web: lanbox.WebFS(), Log: logger})
	httpSrv := &http.Server{
		Addr:              cfg.Listen,
		Handler:           srv,
		ReadHeaderTimeout: 20 * time.Second,
		IdleTimeout:       120 * time.Second,
		TLSConfig:         &tls.Config{MinVersion: tls.VersionTLS12},
	}

	printAccessURLs(cfg.Listen, logger)

	errCh := make(chan error, 1)
	go func() {
		if err := httpSrv.ListenAndServeTLS(certPath, keyPath); err != nil && err != http.ErrServerClosed {
			errCh <- err
			return
		}
		errCh <- nil
	}()

	sigCh := make(chan os.Signal, 1)
	signal.Notify(sigCh, syscall.SIGINT, syscall.SIGTERM)

	select {
	case err := <-errCh:
		return err
	case sig := <-sigCh:
		logger.Printf("info: 收到信号 %v，开始优雅退出", sig)
	}

	srv.Close() // 先断 SSE，避免 Shutdown 等待长连接
	ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
	defer cancel()
	if err := httpSrv.Shutdown(ctx); err != nil {
		return fmt.Errorf("关闭服务超时: %w", err)
	}
	logger.Println("info: 已退出")
	return nil
}

func printPasswordBox(title, password string) {
	line := strings.Repeat("=", 46)
	fmt.Printf("\n%s\n  %s\n  用户名: admin\n  密  码: %s\n  请立即保存，此密码只显示这一次。\n  忘记可用 -reset-admin-password 重置。\n%s\n\n",
		line, title, password, line)
}

func printAccessURLs(listen string, logger *log.Logger) {
	_, port, err := net.SplitHostPort(listen)
	if err != nil {
		port = "8443"
	}
	ips, err := tlsutil.LocalIPs()
	if err != nil || len(ips) == 0 {
		logger.Printf("info: 监听 %s（HTTPS）", listen)
		return
	}
	logger.Printf("info: 监听 %s（HTTPS），可用地址：", listen)
	for _, ip := range ips {
		host := ip
		if strings.Contains(ip, ":") {
			host = "[" + ip + "]"
		}
		logger.Printf("  https://%s:%s", host, port)
	}
}

// 统一的 API 封装：把服务端错误体转换成带 code 的 ApiError。
export class ApiError extends Error {
  constructor(status, code, message) {
    super(message);
    this.status = status;
    this.code = code;
  }
}

async function request(method, url, body) {
  const init = { method, credentials: 'same-origin', headers: {} };
  if (body !== undefined) {
    init.headers['Content-Type'] = 'application/json';
    init.body = JSON.stringify(body);
  }
  const resp = await fetch(url, init);
  if (resp.status === 204) return null;
  const text = await resp.text();
  let data = null;
  if (text) {
    try { data = JSON.parse(text); } catch { /* 非 JSON */ }
  }
  if (!resp.ok) {
    const err = data && data.error ? data.error : { code: 'unknown', message: text || resp.statusText };
    throw new ApiError(resp.status, err.code, err.message);
  }
  return data;
}

export const api = {
  me: () => request('GET', '/api/me'),
  login: (username, password) => request('POST', '/api/login', { username, password }),
  logout: () => request('POST', '/api/logout'),

  list: ({ limit = 50, before = '', q = '', filter = 'all' } = {}) => {
    const p = new URLSearchParams({ limit: String(limit), filter });
    if (before) p.set('before', before);
    if (q) p.set('q', q);
    return request('GET', '/api/items?' + p.toString());
  },
  getItem: (id) => request('GET', '/api/items/' + encodeURIComponent(id)),
  setTaken: (id, taken) => request('PATCH', '/api/items/' + encodeURIComponent(id), { taken }),
  remove: (id) => request('DELETE', '/api/items/' + encodeURIComponent(id)),
  removeFile: (id, fid) =>
    request('DELETE', '/api/items/' + encodeURIComponent(id) + '/files/' + encodeURIComponent(fid)),

  users: () => request('GET', '/api/users'),
  createUser: (username, password, isAdmin) =>
    request('POST', '/api/users', { username, password, is_admin: isAdmin }),
  setUserPassword: (name, password) =>
    request('POST', '/api/users/' + encodeURIComponent(name) + '/password', { password }),
  deleteUser: (name) => request('DELETE', '/api/users/' + encodeURIComponent(name)),
};

// uploadItem 用 XHR 上传以获得真实进度（fetch 没有上传进度）。
export function uploadItem({ text, files, onProgress }) {
  return new Promise((resolve, reject) => {
    const fd = new FormData();
    if (text) fd.append('text', text);
    for (const f of files) fd.append('files', f, f.name);

    const xhr = new XMLHttpRequest();
    xhr.open('POST', '/api/items');
    xhr.withCredentials = true;
    xhr.upload.onprogress = (e) => {
      if (e.lengthComputable && onProgress) onProgress(e.loaded, e.total);
    };
    xhr.onload = () => {
      let data = null;
      try { data = JSON.parse(xhr.responseText); } catch { /* 忽略 */ }
      if (xhr.status >= 200 && xhr.status < 300 && data) {
        resolve(data);
        return;
      }
      const err = data && data.error ? data.error : { code: 'unknown', message: xhr.statusText };
      reject(new ApiError(xhr.status, err.code, err.message));
    };
    xhr.onerror = () => reject(new ApiError(0, 'network', '网络错误，上传中断'));
    xhr.send(fd);
  });
}

export function fmtBytes(n) {
  if (n < 1024) return n + ' B';
  const units = ['KB', 'MB', 'GB', 'TB'];
  let v = n / 1024;
  let i = 0;
  while (v >= 1024 && i < units.length - 1) { v /= 1024; i++; }
  return v.toFixed(v >= 10 ? 0 : 1) + ' ' + units[i];
}

export function fmtTime(iso) {
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return iso;
  const pad = (x) => String(x).padStart(2, '0');
  const today = new Date();
  const sameDay = d.toDateString() === today.toDateString();
  const hm = pad(d.getHours()) + ':' + pad(d.getMinutes());
  if (sameDay) return '今天 ' + hm;
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())} ${hm}`;
}

// EventSource 包装：把事件转给回调，并上报连接状态。
export function connectEvents({ onEvent, onState }) {
  let es = null;
  let closed = false;

  function open() {
    if (closed) return;
    onState('connecting');
    es = new EventSource('/api/events', { withCredentials: true });

    es.addEventListener('open', () => onState('open'));
    es.addEventListener('item.created', (e) => onEvent('item.created', e.data));
    es.addEventListener('item.updated', (e) => onEvent('item.updated', e.data));
    es.addEventListener('item.deleted', (e) => onEvent('item.deleted', e.data));

    es.addEventListener('error', () => {
      onState('error');
      // EventSource 会自动重连；若被关闭（如 401），手动重建
      if (es && es.readyState === EventSource.CLOSED) {
        es.close();
        setTimeout(open, 3000);
      }
    });
  }

  open();
  return {
    close() {
      closed = true;
      if (es) es.close();
    },
  };
}

`README.md`：
```markdown
# lanbox

一个跑在 Linux 小主机上的**文件与消息共享服务**：任一端打开网页发布文字和文件，另一端打开同一个网页取走，反之亦然。两端权限对等，都能发布、查看、下载、标记已取走、删除。

- 单个静态二进制，**零第三方依赖**，无需 Docker、数据库、Node
- HTTPS 单端口，未配置证书时自动生成自签证书
- 每条内容标注**发布者与来源 IP**
- 附件流式落盘（内存占用与文件大小无关），支持 Range 断点下载
- 实时刷新（SSE），另一端发布后 1 秒内出现
- 独立账号 + 会话过期 + 登录限速

## 快速开始

```bash
# 1. 编译（需要 Go 1.26+）
CGO_ENABLED=0 go build -trimpath -ldflags="-s -w" -o lanbox ./cmd/lanbox

# 2. 直接运行（默认监听 :8443，数据在 ./data）
./lanbox

# 3. 首次启动会打印 admin 的随机密码，请立即保存
#    https://<小主机IP>:8443
```

浏览器首次访问自签证书会警告，选择"继续访问"；需要长期免警告见下方"信任自签证书"。

## 配置

启动时读取 `-config` 指定的文件（默认 `config.json`），**所有字段可省略**：

```json
{
  "listen": ":8443",
  "data_dir": "./data",
  "tls": { "cert_file": "", "key_file": "" },
  "max_file_size_mb": 1024,
  "min_free_disk_mb": 1024,
  "max_text_bytes": 65536,
  "session_ttl_hours": 168
}
```

| 字段 | 默认 | 说明 |
|---|---|---|
| `listen` | `:8443` | 监听地址。只对某一网卡开放可写 `192.168.1.9:8443` |
| `data_dir` | `./data` | 数据目录（内容、用户库、证书） |
| `tls.cert_file` / `tls.key_file` | 空 | 同时填写则使用你自己的证书；留空则自动生成自签证书 |
| `max_file_size_mb` | 1024 | 单文件上限，超出返回 413 |
| `min_free_disk_mb` | 1024 | 剩余磁盘保护线，低于此值拒绝上传（507） |
| `max_text_bytes` | 65536 | 单条文字上限 |
| `session_ttl_hours` | 168 | 会话有效期（滑动续期） |

也可用环境变量覆盖：`LANBOX_LISTEN`、`LANBOX_DATA_DIR`、`LANBOX_TLS_CERT_FILE`、`LANBOX_TLS_KEY_FILE`、`LANBOX_MAX_FILE_SIZE_MB`、`LANBOX_MIN_FREE_DISK_MB`、`LANBOX_MAX_TEXT_BYTES`、`LANBOX_SESSION_TTL_HOURS`。

## 账号

- 首次启动自动创建 `admin`，随机 16 位密码打印到 stdout（systemd 下见 `journalctl -u lanbox`）
- 忘记密码：`./lanbox -reset-admin-password`（会打印新的随机密码）
- 管理员可在网页"用户管理"里新增/删除用户、重置密码
- 所有用户权限对等（都能发布/下载/删除）；管理员只是多了用户管理

## 数据与备份

```
data/
├── items/<内容ID>/meta.json    # 元数据
├── items/<内容ID>/files/…       # 附件
├── users.json                  # 账号（PBKDF2 哈希）

备份整个 `data/` 即可；删除某条内容 = 删除对应目录（服务运行时请通过网页删除）。

## 信任自签证书

浏览器：访问一次 → "高级" → "继续前往"。
Linux 客户端：
```bash

sudo update-ca-certificates
```
Windows：把 `cert.pem` 重命名为 `cert.crt` 后"导入到受信任的根证书颁发机构"。

## 安全提示

- 服务只应暴露在你控制的网络内，并保持系统与证书更新
- 真实证书：`tls.cert_file` / `tls.key_file` 指向你的证书即可
- 会话令牌仅存内存，重启服务后需重新登录
- 上传的 HTML/SVG 一律以 `attachment` + `application/octet-stream` 下载，不会在浏览器中执行
